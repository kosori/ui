---
title: Calendar
description: A date picker component with range and multi-select support.
---

Calendar wraps [@daypicker/react](https://daypicker.dev/) with kosori's styling,
supports single, multiple and range selection, and marks itself with
`data-slot`.

## Installation

```bash
npx shadcn@latest add @kosori/calendar
```

## Example

<ComponentPreview name='calendar' />

## Usage

```tsx
'use client';

import { useState } from 'react';
import { Calendar } from '@/components/ui/calendar';

const CalendarDemo = () => {
  const [date, setDate] = useState<Date | undefined>();

  return <Calendar mode='single' onSelect={setDate} selected={date} />;
};
```
